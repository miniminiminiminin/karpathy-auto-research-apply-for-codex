# Failure Memory Activation Checklist

PASS IF
  same_command_or_tool_failure_happened_more_than_once
  OR second_occurrence_of_the_same_debugging_or_verification_workaround
  OR repeated_root_cause_investigation_gap_or_guessing_pattern
  OR workaround_changed_future_behavior
  OR deploy_or_verification_quirk_is_easy_to_forget
  OR operator_is_about_to_move_on_without_capturing_the_lesson

FAIL IF
  event_is_only_routine_progress
  OR workaround_has_no_future_behavior_change

AND before_leaving_the_task:
  - log_the_lesson
  - decide_if_it_is_reusable
  - same_session_promotion_expected_when_repo_behavior_should_change
  - promote_reusable_behavior_in_the_same_session_when_possible
