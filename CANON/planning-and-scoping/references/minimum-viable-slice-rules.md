# Minimum Viable Slice Rules

```text
USE_THIS_REFERENCE IF
  the_team_can_name_many_good_ideas_but_not_the_first_owned_slice
  OR the_plan_keeps_widening
  OR reuse_vs_rebuild_is_still_implicit

MINIMUM_USEFUL_SLICE := PASS IF
  the_slice_delivers_one_real_outcome_or_proof_bearing_unlock
  AND smaller_alternatives_were_considered_and_rejected_explicitly
  AND later_improvements_can_be_parked_without_blocking_this_slice
  AND the_slice_has_one_main_proof_path

DO_LESS := PASS IF
  existing_assets_or_paths_are_checked_before_new_work_is_added
  AND the_plan_prefers_irreducible_core_over_complete_system_fantasy
  AND nice_to_have_polish_is_parked_not_blended_into_the_active_step

ANTI_REDUNDANCY := PASS IF
  two_steps_do_not_prove_the_same_thing
  AND one_owned_file_or_surface_does_not_have_two_competing_active_owners
  AND decomposition_reduces_guesswork_instead_of_cloning_work

FAIL IF
  the_first_slice_is_already_a_program_of_work
  OR the_current_step_exists_only_because_the_goal_is_still_blurry
  OR reuse_before_rebuild_was_never_checked
  OR the_plan_keeps_multiple parallel proof paths under_one_active_step
```
