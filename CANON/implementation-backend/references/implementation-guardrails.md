# Backend Implementation Guardrails

```text
GUARDRAILS := PASS IF
  contract_and_boundary_behavior_are_defined_before_implementation_detail
  AND the_irreducible_core_of_the_seam_is_named_before_new_mechanism_is_added
  AND existing_platform_or_shared_capability_is_reused_before_bespoke_infrastructure_is_added
  AND side_effects_are_isolated_at_explicit_runtime_edges_or_adapters
  AND each_class_or_module_has_one_primary_runtime_responsibility
  AND transport_policy_domain_logic_and_persistence_are_not_collapsed_without_reason
  AND file_growth_past_about_200_lines_requires_an_irreducible_seam_reason_or_a_split
  AND null_empty_timeout_duplicate_and_partial_failure_cases_are_treated_as_first_class_work

FAIL IF
  a_new_mechanism_exists_without_a_reuse_or_irreducible_core_reason
  OR a_class_or_module_became_a_god_object
  OR a_file_grew_past_about_200_lines_without_a_named_irreducible_reason
  OR delivery_logic_widens_into_architecture
  OR migration_strategy_is_required_but_missing
  OR cross_consumer_redesign_is_smuggled_into_implementation
```
