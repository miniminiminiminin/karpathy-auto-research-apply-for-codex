# Component Implementation Playbook

```text
AUTHORITY := {
  type := mixed_execution_guide_and_house_style,
  note := "Semantic structure, state clarity, and accessibility are normative. Several visual hierarchy heuristics are local policy."
}

PREPARE_SEAM := {
  state_component_purpose_in_one_sentence,
  name_framework_or_runtime_expectation,
  name_props_or_input_contract,
  define_typing_strategy,
  choose_styling_approach,
  decide_tokens_variants_or_composition,
  decide_error_boundary_or_parent_fallback,
  list_required_states_before_markup
}

PRESENTATION_SCOPE := PASS IF
  component_only_presentation_work_is_explicit_when_the_seam_is_not_a_full_page
  AND single_component_surface_is_kept_free_of_unrelated_copy_or_layout_scaffolding
  AND mock_data_is_used_without_hiding_real_state_edges
  AND screenshots_or_media_are_cropped_or_simplified_when_full_fidelity_assets_break_clarity
  AND preview_safe_markup_is_produced_when_the_component_must_be_reviewed_in_isolation

BUILD_ORDER :=
  1 -> semantic_structure
  2 -> variant_and_token_wiring
  3 -> interaction_and_focus_behavior
  4 -> loading_empty_error_and_blocked_states
  5 -> responsive_layout_adjustments
  6 -> targeted_tests_or_rendering_proof

GUARDRAILS := PASS IF
  variant_expansion_is_preferred_over_local_style_exceptions
  AND props_do_not_only_mirror_css_toggles_without_user_meaning
  AND interaction_state_and_accessibility_state_stay_in_sync
  AND isolated_preview_output_still_uses_real_tokens_and_responsive_constraints
  AND usage_notes_exist_when_component_has_sharp_edges
  AND prop_or_interface_contract_exists_before_implementation_for_react_seams
  AND unit_test_seam_is_clear_for_state_interaction_and_accessibility_behavior
  AND primary_secondary_and_tertiary_actions_are_visually_distinct_when_the_component_exposes_multiple_actions
  AND secondary_and_tertiary_actions_are_softened_before_primary_actions_gain_more_decoration
  AND visual_weight_is_compensated_before_size_is_reduced
  AND mixed_size_text_aligns_by_baseline_when_read_together
  AND labels_are_subordinate_to_values_when_scan_priority_demands_it
  AND semantic_state_feedback_prefers_tinted_surfaces_with_same_hue_text_icons_or_badges_before_full_strength_fill_blocks
  AND default_elements_can_be_upgraded_without_breaking_semantic_structure
  AND user_supplied_media_is_contained_with_fixed_shapes_or_crop_rules
  AND overlay_strategy_is_defined_before_text_is_placed_on_variable_imagery
  AND text_on_images_requires_overlay_or_contrast_normalization_before_shipping
  AND hero_or_banner_images_use_a_single_contrast_strategy_across_the_full_text_zone_instead_of_patchy_local_fixes
  AND background_image_treatments_change_image_energy_before_foreground_chrome_is_restyled
  AND empty_state_variants_remove_inactive_controls_when_those_controls_have_no_effect
  AND blocked_or_inactive_controls_are_removed_or_demoted_when_user_progress_would_otherwise_stall
  AND empty_state_layouts_remove_or_demote_surrounding_scaffolding_when_that_scaffolding_competes_with_the_recovery_action
  AND empty_state_illustration_or_supporting_media_is_kept_secondary_to_headline_explanation_and_primary_action
  AND component_conventions_can_be_reinterpreted_when_hierarchy_or_comprehension_improves
  AND overlap_is_reserved_for_real_surface_transitions_or_focus_moments_not_for_random_internal_collisions
  AND overlapping_images_or_cards_keep_background_colored_buffer_space_so_touching_edges_do_not_clash
  AND component_overrides_prefer_spacing_background_or_shadow_before_extra_borders
  AND accent_borders_are_sparse_supporting_signals_not_the_main_structure_unless_brand_or_data_encoding_requires_otherwise
```
